cmake_minimum_required(VERSION 3.22)
project(ImageOptimJXL C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature TOOLS DEVTOOLS DOXYGEN MANPAGES BENCHMARK EXAMPLES JNI SJPEG OPENEXR VIEWERS TCMALLOC PLUGINS)
    set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
set(JPEGXL_FORCE_SYSTEM_HWY OFF CACHE BOOL "" FORCE)
set(JPEGXL_FORCE_SYSTEM_BROTLI OFF CACHE BOOL "" FORCE)
# Require the pinned sources; never fall back to machine-local codec libraries.
foreach(dependency brotli highway skcms)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/third_party/${dependency}/LICENSE")
        message(FATAL_ERROR "Initialize jxl/src submodules before building (${dependency} missing)")
    endif()
endforeach()
add_subdirectory(src EXCLUDE_FROM_ALL)
add_executable(jxloptim jxloptim.cc)
target_link_libraries(jxloptim PRIVATE jxl)
option(IMAGEOPTIM_JXL_TESTS "Build JPEG XL optimizer checks" OFF)
if(IMAGEOPTIM_JXL_TESTS)
    enable_testing()
    add_executable(jxloptim-test test.cc)
    target_link_libraries(jxloptim-test PRIVATE jxl)
    target_compile_options(jxloptim-test PRIVATE -UNDEBUG)
    add_test(NAME jxloptim COMMAND jxloptim-test)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/jxl-LICENSE" "JPEG XL and bundled dependencies\n")
foreach(license src/LICENSE src/PATENTS src/third_party/brotli/LICENSE src/third_party/highway/LICENSE src/third_party/skcms/LICENSE)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${license}" contents)
    file(APPEND "${CMAKE_BINARY_DIR}/jxl-LICENSE" "\n${license}\n${contents}\n")
endforeach()
