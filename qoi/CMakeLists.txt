cmake_minimum_required(VERSION 3.22)
project(ImageOptimQOI CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(qoioptim qoioptim.cc)
option(IMAGEOPTIM_QOI_TESTS "Build QOI optimizer checks" OFF)
if(IMAGEOPTIM_QOI_TESTS)
    enable_testing()
    add_executable(qoioptim-test test.cc)
    target_compile_options(qoioptim-test PRIVATE -UNDEBUG)
    add_test(NAME qoioptim COMMAND qoioptim-test)
endif()
configure_file(vendor/LICENSE qoi-LICENSE COPYONLY)
