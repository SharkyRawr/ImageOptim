cmake_minimum_required(VERSION 3.22)
project(ImageOptimAVIF C CXX)
include(FetchContent)
if(AOM_TARGET_CPU STREQUAL "x86_64")
    find_program(IMAGEOPTIM_NASM nasm HINTS /opt/homebrew/bin /usr/local/bin REQUIRED)
    configure_file(nasm.sh.in nasm @ONLY)
    set(CMAKE_ASM_NASM_COMPILER "${CMAKE_CURRENT_BINARY_DIR}/nasm" CACHE FILEPATH "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# libavif configures libaom locally; this override uses our pinned submodule without downloads.
set(FETCHCONTENT_SOURCE_DIR_LIBAOM "${CMAKE_CURRENT_SOURCE_DIR}/aom")
set(AVIF_CODEC_AOM LOCAL CACHE STRING "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libavif SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src")
FetchContent_MakeAvailable(libavif)
add_executable(avifoptim avifoptim.c)
target_link_libraries(avifoptim PRIVATE avif)
option(IMAGEOPTIM_AVIF_TESTS "Build AVIF optimizer checks" OFF)
if(IMAGEOPTIM_AVIF_TESTS)
    enable_testing()
    add_executable(avifoptim-test test.c)
    target_link_libraries(avifoptim-test PRIVATE avif)
    target_compile_options(avifoptim-test PRIVATE -UNDEBUG)
    target_compile_definitions(avifoptim-test PRIVATE TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/src/tests/data")
    add_test(NAME avifoptim COMMAND avifoptim-test)
endif()
file(READ "${libavif_SOURCE_DIR}/LICENSE" avif_license)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/aom/LICENSE" aom_license)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/aom/PATENTS" aom_patents)
file(WRITE "${CMAKE_BINARY_DIR}/avif-LICENSE" "libavif\n${avif_license}\nlibaom\n${aom_license}\n${aom_patents}")
